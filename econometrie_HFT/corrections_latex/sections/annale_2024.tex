\begin{questionbox}{Q1 --- CAPM (espérance et variance)}
Give the CAPM equations for expectation and variance of returns, providing accurate definitions of all quantities. What does the CAPM tell us about the relationship between risk and returns in finance? How can you prove the validity of CAPM in practice? (give main mathematical elements without entering in details)
\end{questionbox}

\begin{solutionbox}{Q1}
\textbf{Notations.} $R_i$ rendement de l'actif $i$, $R_M$ rendement du portefeuille de marché, $r_f$ taux sans risque sur la période, $\beta_i=\Cov(R_i,R_M)/\Var(R_M)$.

\medskip
\textbf{Équation d'espérance (Security Market Line).}
\[
\E[R_i] = r_f + \beta_i\big(\E[R_M]-r_f\big).
\]
\textbf{Interprétation :} prime de risque proportionnelle au \textbf{risque systématique} $\beta_i$.

\medskip
\textbf{Décomposition de variance (sous modèle de marché).} Si
\[
R_i-r_f = \alpha_i + \beta_i(R_M-r_f)+\varepsilon_i,\qquad \E[\varepsilon_i]=0,\ \Cov(\varepsilon_i,R_M)=0,
\]
alors
\[
\Var(R_i) = \beta_i^2\Var(R_M)+\Var(\varepsilon_i).
\]
La partie $\beta_i^2\Var(R_M)$ est le risque \textbf{non diversifiable}; $\Var(\varepsilon_i)$ est diversifiable.

\medskip
\textbf{Lien risque--rendement (message CAPM).}
\begin{itemize}[leftmargin=*]
  \item Les investisseurs sont rémunérés uniquement pour le risque non diversifiable.
  \item Deux actifs avec même $\beta$ doivent offrir la même espérance (sinon arbitrage via portefeuille factoriel).
\end{itemize}

\medskip
\textbf{Validation empirique (éléments mathématiques).}
\begin{itemize}[leftmargin=*]
  \item \textbf{Régressions time-series} : estimer $\beta_i$ et tester $\alpha_i=0$ (GRS).
  \item \textbf{Fama--MacBeth} : tester la relation linéaire $\E[R_i-r_f]=\lambda \beta_i$ en coupe transversale.
  \item \textbf{Problème d'identification} : le vrai marché est inobservable (proxy), $\beta$ bruitée, instabilité de paramètres.
\end{itemize}
\end{solutionbox}

\begin{questionbox}{Q2 --- PCA sur courbes de taux}
You have monthly interest rate curves (2000--2023), 10 maturities (1y--10y). Describe how you would analyze PCA results and what you expect to see.
\end{questionbox}

\begin{solutionbox}{Q2}
\textbf{Données.} Matrice $X$ (dates $\times$ maturités). On travaille souvent sur :
\begin{itemize}[leftmargin=*]
  \item niveaux $y_t(\tau)$ (yields), ou
  \item variations $\Delta y_t(\tau)$ (plus stationnaire),
\end{itemize}
après centrage (et parfois standardisation).

\medskip
\textbf{PCA.} Calculer covariance (ou corrélation) empirique sur les 10 maturités, puis décomposer en valeurs/vecteurs propres. Analyser :
\begin{itemize}[leftmargin=*]
  \item \textbf{Variance expliquée} par les 1--3 premières composantes (scree plot).
  \item \textbf{Loadings} (vecteurs propres) selon la maturité.
  \item \textbf{Scores} (facteurs) dans le temps et leur interprétation économique.
\end{itemize}

\medskip
\textbf{Ce qu'on attend typiquement (résultat classique).}
\begin{itemize}[leftmargin=*]
  \item PC1 : \textbf{Level} (toutes maturités bougent dans le même sens) $\Rightarrow$ loadings \(\approx\) constants positifs.
  \item PC2 : \textbf{Slope} (court vs long) $\Rightarrow$ loadings négatifs sur court, positifs sur long (ou inverse).
  \item PC3 : \textbf{Curvature} (milieu vs extrêmes) $\Rightarrow$ loadings en ``bosse''.
\end{itemize}
Les 3 premières PC expliquent souvent \(\gtrsim 95\%\) de la variance des variations de taux.

\medskip
\textbf{Diagnostics à mentionner.}
\begin{itemize}[leftmargin=*]
  \item stabilité des loadings (rolling PCA),
  \item impact de l'usage niveaux vs variations,
  \item robustesse à la standardisation,
  \item interprétation macro (politique monétaire, term premium).
\end{itemize}
\end{solutionbox}

\begin{questionbox}{Q3 --- Corrélation de deux Itô}
Two Itô processes $X,Y$ driven by correlated Brownian motions with constant correlation $\rho$ and stochastic volatilities. Observed on $[0,1]$ at $i/n$. Give an estimator of $\rho$ and sketch a proof of consistency as $n\to\infty$.
\end{questionbox}

\begin{solutionbox}{Q3}
\textbf{Modèle (forme standard).}
\[
dX_t=\mu^X_t\,dt+\sigma^X_t\,dW_t,\qquad
dY_t=\mu^Y_t\,dt+\sigma^Y_t\,dB_t,\qquad
d\langle W,B\rangle_t=\rho\,dt.
\]

\medskip
\textbf{Estimateur naturel : corrélation réalisée.} Incréments $\Delta_i^n X=X_{i/n}-X_{(i-1)/n}$, idem $Y$.
\[
\widehat{[X,Y]}_n := \sum_{i=1}^n \Delta_i^n X\,\Delta_i^n Y,\qquad
\widehat{[X]}_n := \sum_{i=1}^n (\Delta_i^n X)^2,\qquad
\widehat{[Y]}_n := \sum_{i=1}^n (\Delta_i^n Y)^2.
\]
Puis
\[
\hat\rho_n := \frac{\widehat{[X,Y]}_n}{\sqrt{\widehat{[X]}_n\,\widehat{[Y]}_n}}.
\]

\medskip
\textbf{Pourquoi c'est consistant (esquisse).}
\begin{itemize}[leftmargin=*]
  \item Pour des semimartingales continues, \textbf{variation quadratique} :
  \[
  \widehat{[X]}_n \xrightarrow[n\to\infty]{\mathbb{P}} [X]_1=\int_0^1 (\sigma^X_t)^2\,dt,\qquad
  \widehat{[Y]}_n \to [Y]_1=\int_0^1 (\sigma^Y_t)^2\,dt.
  \]
  \item \textbf{Covariation quadratique} :
  \[
  \widehat{[X,Y]}_n \xrightarrow{\mathbb{P}} [X,Y]_1=\int_0^1 \sigma^X_t\sigma^Y_t\,d\langle W,B\rangle_t
  =\rho\int_0^1 \sigma^X_t\sigma^Y_t\,dt.
  \]
  \item Donc, si les intégrales au dénominateur sont $>0$, alors
  \[
  \hat\rho_n \xrightarrow{\mathbb{P}} \rho\cdot
  \frac{\int_0^1 \sigma^X_t\sigma^Y_t\,dt}{\sqrt{\int_0^1(\sigma^X_t)^2dt\ \int_0^1(\sigma^Y_t)^2dt}}.
  \]
  \textbf{Remarque importante :} cet estimateur converge vers la \emph{corrélation intégrée} si les volatilitès sont stochastiques. Pour récupérer la \textbf{corrélation instantanée constante} $\rho$, il faut supposer (ou utiliser une méthode) garantissant que le facteur d'échelle vaut 1 (par ex.\ $\sigma^X_t=\sigma^Y_t$ ou normalisation/estimations locales).
\end{itemize}

\medskip
\textbf{Version ``qui identifie $\rho$'' sous hypothèse standard cours.} Souvent, on suppose un modèle à deux Brownien corrélés :
\[
dX_t=\sigma^X_t\,dW_t,\qquad dY_t=\sigma^Y_t(\rho\,dW_t+\sqrt{1-\rho^2}\,dW_t^\perp),
\]
alors $[X,Y]_1=\rho\int_0^1\sigma^X_t\sigma^Y_t\,dt$ et l'argument précédent donne la consistance de l'estimateur vers $\rho$ si la normalisation est discutée dans le cours (corrélation ``au niveau Brownien'').
\end{solutionbox}

\begin{questionbox}{Q4 --- Rough volatility}
How would you demonstrate that rough volatility models are superior to conventional stochastic volatility models? (1 page maximum, several answers are possible)
\end{questionbox}

\begin{solutionbox}{Q4}
\selectlanguage{english}
A convincing answer combines \textbf{empirical evidence} (roughness in time series), \textbf{option evidence} (better fit/dynamics), and \textbf{economic consistency} (no-arbitrage and microstructure foundations).

\medskip
\textbf{1) Empirical evidence from high-frequency data (roughness).} Build a volatility proxy (realized variance/volatility) and test the scaling of log-volatility increments. Let \(X_t=\log RV_{t,\Delta}\). Estimate the variogram
\[
\widehat{\mathcal V}(h)=\frac{1}{T_h}\sum_t (X_{t+h}-X_t)^2.
\]
Rough volatility predicts \(\mathcal V(h)\approx C h^{2H}\) with \(H\in(0,1/2)\). A log--log regression of \(\widehat{\mathcal V}(h)\) on \(h\) typically yields \(H\approx 0.05\)--\(0.2\), consistently across many assets. This is difficult to reconcile with classical smooth Markov diffusions for volatility, whose small-lag behaviour corresponds to \(H\approx 1/2\) (too smooth).

\medskip
\textbf{2) Option evidence (fit and dynamics).} Compare rough models (e.g.\ rBergomi/rough Heston) against standard models (Heston/SABR) on:
\begin{itemize}[leftmargin=*]
  \item \textbf{Cross-sectional fit.} Pricing errors across strikes and maturities, with the same number of effective parameters.
  \item \textbf{Short-maturity behaviour.} Rough models reproduce the observed steep short-end skew term structure better than classical SV (often labelled ``skew explosion'' phenomena).
  \item \textbf{Dynamic performance.} Out-of-sample re-calibration stability and hedging error (delta/vega hedging) on a rolling window.
\end{itemize}

\medskip
\textbf{3) Link to arbitrage (model admissibility).} A fair comparison requires both models to be \textbf{no-arbitrage} and calibrated consistently under the risk-neutral measure. Rough models can be built as Volterra volatility models that preserve positivity and admit well-posed pricing/hedging. The claim is not ``roughness creates arbitrage''; rather, roughness is a statistically necessary feature that can be embedded in an arbitrage-free pricing framework.

\medskip
\textbf{4) Microstructure/impact foundations (high-value connection to the course).} Order flow exhibits clustering (Hawkes/self-excitation) and prices respond to flow through transient impact. In near-critical regimes, scaling limits of Hawkes-driven activity combined with impact kernels lead to Volterra/fractional-type volatility, providing a microstructural route to \(H<1/2\). This gives rough volatility an economic foundation instead of a purely phenomenological fit.

\medskip
\textbf{Conclusion.} Rough volatility is ``superior'' if it (i) matches the empirical scaling of volatility, (ii) improves option surface fit/dynamics and hedging out-of-sample, and (iii) remains consistent with no-arbitrage and with a plausible microstructure mechanism.
\selectlanguage{french}
\end{solutionbox}

\begin{questionbox}{Q5 --- Hawkes, impact, rough}
Using Hawkes processes, explain the connection between market microstructure, market impact and rough volatility (1 page maximum).
\end{questionbox}

\begin{solutionbox}{Q5}
\selectlanguage{english}
\textbf{Microstructure $\rightarrow$ impact $\rightarrow$ rough volatility} can be explained in three steps.

\medskip
\textbf{1) Microstructure: order flow is self-exciting.} A Hawkes process models the clustering of market activity:
\[
\lambda_t=\mu+\int_0^t \phi(t-s)\,dN_s,\qquad \phi\ge 0,
\]
where \(N_t\) counts events (trades, market orders, cancellations) and \(\phi\) measures self-excitation. The branching ratio
\[
m=\int_0^\infty \phi(u)\,du
\]
quantifies endogeneity: when \(m\) is close to \(1\) (near-criticality), bursts of activity trigger further activity, producing persistence across many time scales (long memory-like behaviour in intensities).

\medskip
\textbf{2) Market impact: prices respond to signed order flow with memory.} A stylized transient-impact model writes
\[
P_t=P_0+\int_0^t G(t-s)\,dQ_s,
\]
where \(Q\) is signed volume (buy minus sell pressure) and \(G\) is a decaying impact kernel (transient impact). If order flow is clustered (Hawkes), then both \(Q\) and the induced price variation inherit multi-scale dependence. In particular, the quadratic variation of \(P\) (hence realized volatility) becomes driven by a stochastic activity rate.

\medskip
\textbf{3) Scaling limits: near-critical Hawkes $\Rightarrow$ Volterra/fractional limits.} In near-critical regimes, appropriate time rescaling and renormalization lead Hawkes intensities (or integrated intensities) to converge to \textbf{Volterra-type} processes (fractional/rough limits). Because volatility is essentially the rate of quadratic variation, these Volterra limits translate into \textbf{rough} volatility dynamics with Hurst parameter \(H<1/2\).

\medskip
\textbf{Takeaway.} Rough volatility is not only a good reduced-form fit: it can emerge from microstructure primitives (self-exciting order flow) combined with realistic impact (transient kernels). This links high-frequency market mechanics to low-frequency volatility roughness.
\selectlanguage{french}
\end{solutionbox}

\begin{questionbox}{Q6 --- Commentaire d'article}
Summarize and comment the enclosed article in light of the course: results, strengths/weaknesses, limitations, improvements.
\end{questionbox}

\begin{solutionbox}{Q6}
\selectlanguage{english}
This question is graded on structure and critical thinking. A high-scoring answer reads like a short referee report: clear summary, explicit links to course concepts, then targeted criticism and improvements.

\medskip
\textbf{Summary.} The paper addresses a specific market question (prediction, volatility modelling, microstructure, execution, etc.) using a dataset with a given frequency and horizon. It proposes a model/estimation procedure and reports quantitative results (fit, predictability, economic value, or structural estimates).

\medskip
\textbf{Connection to the course.} Place the contribution in one (or more) course frameworks: CAPM/factor models for risk premia; regularization (Ridge/Lasso) for high-dimensional prediction; realized measures and rough volatility for volatility modelling; Hawkes/queue-reactive dynamics for order flow; Almgren--Chriss or transient impact for execution and market impact.

\medskip
\textbf{Strengths.} The strongest papers (i) use a well-identified empirical design, (ii) validate out-of-sample with honest baselines, (iii) report uncertainty (confidence intervals) and stability, and (iv) align metrics with the goal (pricing error for options, IS/market impact for execution, log-likelihood for point processes, etc.).

\medskip
\textbf{Weaknesses and limitations.} Typical issues are endogeneity (especially in impact/execution), microstructure noise and sampling bias at high frequency, non-stationarity and regime shifts, data snooping (multiple testing), and ignoring transaction costs/impact when claiming economic gains.

\medskip
\textbf{Improvements.} Propose 2--4 concrete extensions: add transaction-cost/impact-aware evaluation; use walk-forward validation; compare against stronger baselines; add regime dependence (volatility/liquidity buckets); run time-rescaling tests for point processes; or test predictive gains statistically (Diebold--Mariano). If the paper proposes a strategy, insist on implementability (latency, market impact, capacity).

\medskip
\textbf{Conclusion.} The paper is convincing if its gains remain robust after costs/impact and if its identification assumptions remain credible under microstructure effects and non-stationarity.
\selectlanguage{french}
\end{solutionbox}
